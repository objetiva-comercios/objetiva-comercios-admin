    #!/usr/bin/env bash
    set -e

    echo "Phase 38 — Triple verification (D-04)"
    echo "============================================================"
    echo ""

    # ---- STEP 1: drizzle-kit check + generate --check (no DB needed) ----
    echo "Step 1: Local journal+snapshot consistency"

    pnpm --filter backend exec drizzle-kit generate --check
    echo "✓ drizzle-kit generate --check passed"
    echo ""

    # ---- STEP 2: drizzle-kit pull contra prod + diff vs schema.ts ----
    echo "Step 2: Introspect prod schema and diff vs schema.ts"
    OUT="/tmp/phase38-drift-check-$(date +%Y%m%d-%H%M)"
    mkdir -p "$OUT"

    if [ -z "${DATABASE_URL_PROD:-}" ]; then
      echo "❌ DATABASE_URL_PROD not set. Export it before running this step."
      echo "   Example: export DATABASE_URL_PROD='postgresql://sanchez:PASS@localhost:5432/erp_sanchez'"
      exit 1
    fi

    DATABASE_URL="$DATABASE_URL_PROD" \
      pnpm --filter backend exec drizzle-kit pull \
      --out="$OUT" --dialect=postgresql

    DIFF_FILE="$OUT/schema.diff"
    diff -u apps/backend/src/db/schema.ts "$OUT/schema.ts" > "$DIFF_FILE" || true

    echo "✓ Diff written to $DIFF_FILE"
    echo ""
    echo "Diff summary (real differences only):"
    grep -E "^[+-][^+-]" "$DIFF_FILE" | head -50
    echo ""

    # ---- STEP 3: smoke playwright (executed in Task 3 — separate human checkpoint) ----
    echo "Step 3: smoke playwright must be executed via the playwright-cli skill — see Task 3."
    echo ""

    echo "============================================================"
    echo "✓ Verification steps 1 + 2 complete"
    echo "  Diff file:        $DIFF_FILE"
    echo "  Next:             Task 2 (analyze diff + draft DRIFT-REPORT.md)"
    echo "                    Task 3 (smoke playwright — humano)"
