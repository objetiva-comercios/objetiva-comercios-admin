<!-- Reference implementation (RESEARCH §4) -->

```typescript
// health.service.ts
import { Injectable } from '@nestjs/common'
import { sql } from 'drizzle-orm'
import { DrizzleService } from '../db'

const CRITICAL_TABLES = [
  'business_settings',
  'articulos',
  'existencias',
  'prop_marca',
  'orders',
] as const
type TableName = (typeof CRITICAL_TABLES)[number]

export type TableStatus = { name: TableName; status: 'ok' | 'fail'; error?: string }
export type DbHealth = { ok: boolean; tables: TableStatus[] }

@Injectable()
export class HealthService {
  constructor(private readonly drizzle: DrizzleService) {}

  async checkDb(): Promise<DbHealth> {
    const results = await Promise.all(
      CRITICAL_TABLES.map(async (name): Promise<TableStatus> => {
        try {
          await this.drizzle.db.execute(sql`SELECT 1 FROM ${sql.identifier(name)} LIMIT 0`)
          return { name, status: 'ok' }
        } catch (err) {
          return { name, status: 'fail', error: (err as Error).message }
        }
      })
    )
    return { ok: results.every(r => r.status === 'ok'), tables: results }
  }
}
```

## Trust Boundaries

| Boundary                                | Description                                                                                                                                                              |
| --------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| Internet → /api/health/db (vía Traefik) | Endpoint público (sin auth). Cualquiera puede curl-earlo desde Internet. Decidido público porque Docker healthcheck no tiene JWT y agregar uno introduce más superficie. |
| docker healthcheck → loopback :3001     | El curl del healthcheck es local al container (no pasa por Traefik). Sin secret leakage.                                                                                 |
| HealthService → DB (rol sanchez)        | Lecturas SELECT 1 LIMIT 0 — query trivial, no expone data.                                                                                                               |

## STRIDE Threat Register

| Threat ID | Category               | Component                                                 | Disposition | Mitigation Plan                                                                                                                                                                                                                                                                                        |
| --------- | ---------------------- | --------------------------------------------------------- | ----------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| T-38-10   | Information Disclosure | /api/health/db response shape                             | mitigate    | Response contiene nombres de tablas (`business_settings`, `articulos`, etc.) — info no secreta (revelable inferiendo schema.ts en el repo público). NO incluir mensajes de error verbosos en producción si el rol del backend está en `production` (logear pero no devolver `error.message` completo). |
| T-38-11   | Denial of Service      | flood de /api/health/db                                   | mitigate    | El query es `SELECT 1 LIMIT 0` × 5 (parsing-only) — costo casi nulo en DB. Aceptable. Si se vuelve un vector real, agregar rate limit en Traefik (no en Phase 38).                                                                                                                                     |
| T-38-12   | Tampering              | malicious sql.identifier injection                        | mitigate    | El parametro `name` viene de un array literal `as const` (whitelist). NO viene de input externo. `sql.identifier()` provee escapeing como defensa en profundidad.                                                                                                                                      |
| T-38-13   | Spoofing               | falsa señal de health (DB caída pero endpoint reporta ok) | mitigate    | El check ejecuta una query real contra cada tabla; si la DB cae, todas las queries fallan y el endpoint reporta `ok: false`. Test obligatorio: ejecutar `docker compose stop postgres` y verificar que `/api/health/db` reporta 503 (Task 5).                                                          |

    import { Controller, Get, HttpException, HttpStatus } from '@nestjs/common'
