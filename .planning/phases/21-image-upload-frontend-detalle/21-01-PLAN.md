```typescript
export async function uploadLogo(
  type: 'square' | 'rectangular',
  file: File
): Promise<BusinessSettings> {
  const headers = await getAuthHeaders()
  const formData = new FormData()
  formData.append('file', file)
  const response = await fetch(`${API_BASE_URL}/api/settings/logo/${type}`, {
    method: 'POST',
    headers,
    body: formData,
  })
  await throwIfError(response)
  return response.json()
}
```

   - Props: `{ tipo: 'etiqueta' | 'producto', slot: number, url: string | null, articuloCodigo: string, onUpdated: (articulo: Articulo) => void, onPreview: () => void }`

   - HTML5 DnD handlers on the container div:
     - `onDragOver`: `e.preventDefault(); e.stopPropagation(); setIsDraggingOver(true)`
     - `onDragLeave`: only set false if `!e.currentTarget.contains(e.relatedTarget as Node)` — prevents false positives from child elements
     - `onDrop`: `e.preventDefault(); setIsDraggingOver(false); uploadFile(e.dataTransfer.files[0])`
   - `onClick`: if not uploading, trigger `inputRef.current?.click()`
   - Upload function: `setIsUploading(true)` → call `uploadArticuloImagen` → `onUpdated(result)` → toast success → `setIsUploading(false)`. On error: toast destructive with error message.
   - **Visual states per user decision:**
     - Empty slot: `border-2 border-dashed border-muted-foreground/30 aspect-square rounded-sm cursor-pointer` with Plus icon and label text (e.g., "Etiqueta 1", "Producto 3")

     - Uploading: `Skeleton` component pulsing with a `Loader2` spinner centered
     - Filled slot: `<img>` showing thumbnail (prepend `API_BASE_URL` + `getThumbUrl(url)` where `getThumbUrl` replaces `_detail.webp` with `_thumb.webp`), with label overlay at bottom
     - Delete button (filled only): absolute positioned top-right, semi-transparent bg on hover, X icon. Calls `deleteArticuloImagen` → `onUpdated(result)` → toast. Use `e.stopPropagation()` to prevent triggering click/preview.
