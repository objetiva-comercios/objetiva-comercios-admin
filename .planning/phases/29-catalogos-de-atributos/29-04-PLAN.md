```typescript
import 'dotenv/config'
import postgres from 'postgres'

/**
 * Seed idempotente para crear el usuario admin de testing E2E en Supabase.
 *
 * Activación: E2E_SEED=true pnpm --filter @objetiva/backend exec tsx src/db/seed-e2e.ts
 *
 * Crea (o re-actualiza) el usuario `e2e-admin@test.local` con app_metadata.role=admin.
 * NOTA: este script asume que `DATABASE_URL` apunta al Supabase de desarrollo.
 * En producción NO se ejecuta — el guard `process.env.E2E_SEED === 'true'` lo impide.
 */
async function main() {
  if (process.env.E2E_SEED !== 'true') {
    console.log('[seed-e2e] Skipping — E2E_SEED is not set to "true"')
    return
  }

  const url = process.env.SUPABASE_DB_URL ?? process.env.DATABASE_URL
  if (!url) {
    throw new Error('[seed-e2e] SUPABASE_DB_URL o DATABASE_URL deben estar definidos')
  }

  const email = process.env.E2E_ADMIN_EMAIL ?? 'e2e-admin@test.local'
  const password = process.env.E2E_ADMIN_PASSWORD ?? 'e2e-admin-pwd-CHANGE-ME'

  const sql = postgres(url, { max: 1 })

  try {
    const existing = await sql`SELECT id FROM auth.users WHERE email = ${email}`
    if (existing.length > 0) {
      // Asegurar role admin en app_metadata
      await sql`
        UPDATE auth.users
        SET raw_app_meta_data = jsonb_set(
          COALESCE(raw_app_meta_data, '{}'::jsonb),
          '{role}',
          '"admin"',
          true
        ),
        updated_at = NOW()
        WHERE email = ${email}
      `
      console.log(`[seed-e2e] OK: user ${email} already existed; ensured role=admin`)
    } else {
      await sql`
        INSERT INTO auth.users (
          instance_id, id, email, encrypted_password,
          email_confirmed_at, raw_app_meta_data, raw_user_meta_data,
          created_at, updated_at, role, aud
        ) VALUES (
          '00000000-0000-0000-0000-000000000000',
          gen_random_uuid(),
          ${email},
          crypt(${password}, gen_salt('bf')),
          NOW(),
          ${JSON.stringify({ role: 'admin', provider: 'email', providers: ['email'] })}::jsonb,
          '{}'::jsonb,
          NOW(), NOW(), 'authenticated', 'authenticated'
        )
      `
      console.log(`[seed-e2e] OK: created admin user ${email}`)
    }
  } finally {
    await sql.end()
  }
}

main().catch((err) => {
  console.error('[seed-e2e] FAILED:', err)
  process.exit(1)
})
```

```typescript
/**
 * Sugerencia automática de `abrev` desde un nombre de propiedad.
 *
 * Reglas (Pattern 6 — RESEARCH.md líneas 937-983):
 *  1. Trim y NFD normalize
 *  2. Strip diacríticos (combining marks U+0300..U+036F)
 *  3. Uppercase
 *  4. Quedarse solo con [A-Z0-9]
 *  5. Tomar las primeras `takeChars` (default 4)
 *  6. Cap a 8 (defense in depth — el regex CHECK ya garantiza 1..8)
 *
 * NOTA: el usuario puede editar libremente; la sugerencia es solo guía visual.
 *
 * @param nombre - String del nombre del que sugerir.
 * @param takeChars - Cantidad de chars iniciales a tomar (default 4).
 */
export function suggestAbrev(nombre: string, takeChars = 4): string {
  return (nombre ?? '')
    .normalize('NFD')
    // Unicode "Combining Diacritical Marks" block (U+0300 — U+036F) escapado
    // para evitar caracteres invisibles en el source. Equivalente semántico
    // al regex literal /[\u0300-\u036F]/g del research, pero portable a linters/CI.
    .replace(/[\u0300-\u036F]/g, '')
    .toUpperCase()
    .replace(/[^A-Z0-9]/g, '')
    .slice(0, takeChars)
    .slice(0, 8) // hard cap
}
```
