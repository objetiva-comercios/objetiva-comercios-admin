    1. **api.client.ts** — Add 3 functions at the end of the file, following existing patterns:

       ```typescript
       // API Keys types
       export interface ApiKeyItem {
         id: number
         name: string
         prefix: string
         createdAt: string
         lastUsedAt: string | null
       }

       export interface ApiKeyCreated extends ApiKeyItem {
         fullKey: string
       }

       export async function fetchApiKeys(): Promise<ApiKeyItem[]> {
         const res = await fetch(`${API_BASE_URL}/api/api-keys`, { headers: await getAuthHeaders() })
         if (!res.ok) throw new Error('Error al cargar API keys')
         return res.json()
       }

       export async function createApiKey(name: string): Promise<ApiKeyCreated> {
         const res = await fetch(`${API_BASE_URL}/api/api-keys`, {
           method: 'POST',
           headers: await getAuthHeaders(),
           body: JSON.stringify({ name }),
         })
         if (!res.ok) throw new Error('Error al crear API key')
         return res.json()
       }

       export async function revokeApiKey(id: number): Promise<void> {
         const res = await fetch(`${API_BASE_URL}/api/api-keys/${id}`, {
           method: 'DELETE',
           headers: await getAuthHeaders(),
         })
         if (!res.ok) throw new Error('Error al revocar API key')
       }
       ```
