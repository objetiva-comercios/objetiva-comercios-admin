```typescript
async function handleConfirmToggle() {
  if (!articulo) return
  setShowToggleDialog(false)
  try {
    if (articulo.activo) {
      await deleteArticulo(articulo.codigo)
      toast({ title: 'Articulo desactivado', description: `"${articulo.nombre}" fue desactivado.` })
      router.push('/articulos')
    } else {
      const updated = await toggleArticuloActivo(articulo.codigo)
      setArticulo(updated)
      toast({ title: 'Articulo reactivado', description: `"${articulo.nombre}" fue reactivado.` })
    }
  } catch (err) {
    toast({
      title: 'Error al cambiar el estado',
      description: err instanceof Error ? err.message : 'Error desconocido',
      variant: 'destructive',
    })
  }
}
```

IMPORTANT: toast BEFORE router.push — Next.js App Router navigation may unmount the component, so toast must fire first.

Do NOT change: AlertDialog text (stays "Desactivar"/"Reactivar"), button labels, or any other UI element. The change is purely which API endpoint gets called.
